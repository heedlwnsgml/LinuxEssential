#!/bin/bash
# 경로 및 환경 변수 설정BASE_DIR="/root/linux"HOST_NAME=$(hostname -s)
TARGET_DIR="$BASE_DIR/$HOST_NAME"

# 시간 변수 설정
CURRENT_TIME=$(date +%Y%m%d_%H%M%S)
LOG_FILE="$TARGET_DIR/cmdlog_${CURRENT_TIME}.txt"
NOW_EXACT=$(date +"%Y-%m-%d %H")

# 디렉토리 생성
if [ ! -d "$TARGET_DIR" ]; then
    mkdir -p "$TARGET_DIR"
fi

# 과거 파일 삭제 (시 단위 정각 기준)
find "$TARGET_DIR" -type f -name "cmdlog_*.txt" ! -newermt "$NOW_EXACT" -exec rm -f {} +

# 기록 시작
echo "로그 저장 위치: $LOG_FILE"
echo "기록을 마치려면 'exit'를 입력하십시오."
script -q "$LOG_FILE"

# 터미널 특수문자 필터링
sed -i -r 's/\x1B\[[0-9;?]*[a-zA-Z]//g' "$LOG_FILE"
sed -i -r 's/\x1B\][0-9]*;[^\x07]*\x07//g' "$LOG_FILE"
sed -i -r 's/\x1B\][0-9]*;[^\x1B]*\x1B\\//g' "$LOG_FILE"

# 백스페이스 기호 치환
col -bx < "$LOG_FILE" > "${LOG_FILE}.tmp"
mv "${LOG_FILE}.tmp" "$LOG_FILE"

# Git 작업 진행
echo "Git 업로드를 진행합니다."
cd "$BASE_DIR" || exit

git add "$LOG_FILE"
git commit -m "add: cmdlog $CURRENT_TIME ($HOST_NAME)"

# 푸시 전 원격 저장소 최신화 (충돌 방지)
git pull --rebase origin main
